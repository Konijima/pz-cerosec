VERSION = 1,

/* The mod's own sandbox options.
 *
 * Build 42 reads this file itself: zombie.sandbox.CustomSandboxOptions.init()
 * walks every installed mod and looks for
 *   <mod>/<version dir>/media/sandbox-options.txt
 *   <mod>/common/media/sandbox-options.txt
 * taking whichever exists, and hands each option to SandboxOptions through
 * newCustomOption. The version dir for this mod is 42, so this file is at
 * 42/media/sandbox-options.txt -- beside 42/media/lua, not at the mod root.
 * The bytecode is quoted in docs/notes/modules-proofs.md.
 *
 * The names a player reads are in 42/media/lua/shared/Translate/EN/Sandbox.json,
 * keyed "Sandbox_" plus what `translation` says here.
 */

/* Does a door, a window or a light have to be WIRED before a computer can see
 * it? On, which is the mod's own answer: a machine reaches what somebody fitted
 * a module to and nothing else. Off is the world as it was before this option
 * existed -- every door, window, lock and light of the building in /dev, wired
 * or not -- and it is there for a server that was already running that way.
 */
option CeroSec.HardwareRequired
{
    type = boolean,
    default = true,
    page = CeroSec,
    translation = CeroSec_HardwareRequired,
}

/* Is there anything already ON the computers, and on the disks beside them?
 *
 * On, which is the mod's own answer and NOT the world as it was before the
 * option existed: a vanilla computer switched on for the first time comes up
 * with somebody's accounts on it, his files, a motd, a week of log, and
 * sometimes a password to find; a floppy off a shelf sometimes has a program and
 * a README on it. Off is the bare machine this mod shipped with -- two open
 * accounts and an empty disk -- and it is there for a server that wants the
 * computers to be a blank sheet.
 *
 * IT IS THE ONE OPTION OF OURS WHOSE DEFAULT CHANGES A WORLD, and it changes it
 * in one direction only: a machine somebody has already switched on is HIS and is
 * never touched (SCeroSecObject:turnOn prefills only where the state was made a
 * moment ago). What changes is the machines nobody has been to yet, which is what
 * this is for. Said again in docs/CONTENT.md, where a player can read it.
 */
option CeroSec.PrefilledMachines
{
    type = boolean,
    default = true,
    page = CeroSec,
    translation = CeroSec_PrefilledMachines,
}

/* May anybody fit and remove modules on a fixture standing in somebody else's
 * safehouse?
 *
 * OFF, which is the world as it was before this option existed and the default
 * a new option of ours takes (the compatibility contract): the safehouse is no
 * gate at all and any survivor with the box, the tool and the trade may wire the
 * door he is standing at.
 *
 * On, a fixture inside a safehouse takes a module only from somebody that
 * safehouse allows -- its owner, its members, and an admin, which is vanilla's
 * own answer (SafeHouse.playerAllowed reads the capability
 * CanGoInsideSafehouses). It is there for a server where stripping a rival's
 * hardware from his own hallway was the sabotage nobody wanted.
 *
 * It has no effect in single player: there are no safehouses, so no fixture
 * stands in one.
 *
 * The safehouse's own vanilla rule about LOOTING is not consulted either way.
 * Fitting is not looting -- nothing leaves the building -- so this option is the
 * only gate there is.
 */
option CeroSec.SafehouseModules
{
    type = boolean,
    default = false,
    page = CeroSec,
    translation = CeroSec_SafehouseModules,
}

/* Does the mod's own log go to the console? Off, which is the default and the
 * way it was before this option existed: the mod keeps its lines in a ring of
 * two hundred that a dedicated server loses at every restart, and prints none
 * of them. On, every line is also printed, as "CeroSec: <line>", to the game's
 * console -- the server console on a dedicated server, console.txt beside it.
 *
 * What it prints is everything the mod says about itself, which is little in
 * a quiet world: every saved job dropped or refused at a load, every book that
 * came back and how many of its jobs did, every machine switched off for lack
 * of power. It is safe to leave on for a bug hunt. CeroSec.DEBUG in
 * CeroSecDefs.lua is the developer's way to the same lines, a constant a
 * server owner cannot flip; this is the owner's.
 *
 * Read at each line (CeroSec.serverLog), so changing it in the sandbox takes
 * effect at once. Only the value true turns it on.
 */
option CeroSec.ServerLog
{
    type = boolean,
    default = false,
    page = CeroSec,
    translation = CeroSec_ServerLog,
}
