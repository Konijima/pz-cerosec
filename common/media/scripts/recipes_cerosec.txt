module CeroSec
{
    /* The four hardware modules, made at a workbench.

       Every key and every spelling here is copied from the vanilla electrical
       recipes rather than from a wiki:
       media/scripts/generated/recipes/recipes_electrical.txt (the improvised
       flashlight at :100 and the improvised lantern at :128) and
       recipes_traps.txt (:3, the remote controller, which is the electrical
       recipe that makes a SENSOR part and is the nearest thing in the game to
       these four).

       - timedAction = MakingElectrical is the animation vanilla gives every
         electrical recipe there is; category = Electrical is the tab they are
         filed under; Tags = AnySurfaceCraft is what lets the job be done on any
         table rather than only at a named workbench. All three are on every
         recipe in both files above.

       - The tools are asked for by TAG and kept: `tags[base:screwdriver]` and
         `tags[base:pliers]`, which is how vanilla asks for them, so any
         screwdriver in the game answers -- and mode:keep means the job does not
         eat them. `flags[MayDegradeVeryLight]` on the pliers is vanilla's own
         wear on the same tool in the same job.

       - The inputs are vanilla items and every one of them was checked to exist
         in B42 before it was written down (docs/notes/modules-proofs.md, 8):
         Base.ElectronicsScrap, Base.ElectricWire, Base.Screws,
         Base.SmallSheetMetal, Base.ScrapMetal, Base.EngineParts. There is no
         Base.Electronics, no Base.Magnet and no Base.CopperWire in this build,
         which is why none of them is here.

       - SkillRequired is the same level the module asks for to be INSTALLED
         (CeroSecModules.LIST: contact 1, relay 1, strike 2, operator 3). A
         survivor who can fit one can build one.

       - NeedToBeLearn, and the book. These four are LEARNED, from
         CeroSec.WiringGuide -- the Field Wiring Guide, found in the world and
         read (items_cerosec.txt, and the engine does the rest:
         IsoGameCharacter.ReadLiterature walks the item's LearnedRecipes and
         calls learnRecipe on each). The recipe NAMES below are therefore load
         bearing twice over: the guide names them back.

       - AutoLearnAll six levels above SkillRequired, which is vanilla's own
         shape for a recipe a magazine teaches and not a departure from it. The
         rung that wrote these first set it EQUAL to SkillRequired, so an
         electrician who could build one already knew how, and the book would
         have had nothing to teach anybody. Vanilla never does that. It keeps
         both keys and it spreads them:

           MakeImprovisedFlashlight   SkillRequired 1, AutoLearnAny 3
             recipes/recipes_electrical.txt:101-109, taught by
             Base.ElectronicsMag5 (literature.txt:5367-5377)
           MakeRemoteControllerV1     SkillRequired 2, AutoLearnAny 8
             recipes/recipes_traps.txt:3-11, taught by
             Base.ElectronicsMag1 (literature.txt:5307-5320)

         So the skill still gates the CRAFT at the level the module is fitted
         at, the book is how a survivor actually comes by the recipe, and a
         master electrician works it out for himself eventually. Six is the
         remote controller's own gap (2 to 8), and the remote controller is the
         vanilla recipe these four are nearest to.

         `All` rather than vanilla's `Any` is kept from the first rung, so that
         a second skill added here later would tighten the gate and not widen
         it.

       - xpAward is set against the vanilla recipes above, which pay 10 for a
         flashlight and 6 to 14 for a remote controller. */

    craftRecipe MakeCeroSecMagneticContact
    {
        timedAction = MakingElectrical,
        time = 60,
        NeedToBeLearn = true,
        SkillRequired = Electricity:1,
        Tags = AnySurfaceCraft,
        category = Electrical,
        AutoLearnAll = Electricity:7,
        xpAward = Electricity:5,
        inputs
        {
            item 1 tags[base:screwdriver] mode:keep flags[Prop1],
            item 1 [Base.ElectronicsScrap],
            item 1 [Base.ElectricWire],
            item 1 [Base.Screws],
        }
        outputs
        {
            item 1 CeroSec.MagneticContact,
        }
    }

    craftRecipe MakeCeroSecRelay
    {
        timedAction = MakingElectrical,
        time = 80,
        NeedToBeLearn = true,
        SkillRequired = Electricity:1,
        Tags = AnySurfaceCraft,
        category = Electrical,
        AutoLearnAll = Electricity:7,
        xpAward = Electricity:8,
        inputs
        {
            item 1 tags[base:screwdriver] mode:keep flags[Prop1],
            item 1 tags[base:pliers] mode:keep flags[MayDegradeVeryLight],
            item 2 [Base.ElectronicsScrap],
            item 1 [Base.ElectricWire],
        }
        outputs
        {
            item 1 CeroSec.Relay,
        }
    }

    craftRecipe MakeCeroSecElectricStrike
    {
        timedAction = MakingElectrical,
        time = 120,
        NeedToBeLearn = true,
        SkillRequired = Electricity:2,
        Tags = AnySurfaceCraft,
        category = Electrical,
        AutoLearnAll = Electricity:8,
        xpAward = Electricity:10,
        inputs
        {
            item 1 tags[base:screwdriver] mode:keep flags[Prop1],
            item 1 tags[base:pliers] mode:keep flags[MayDegradeVeryLight],
            item 2 [Base.ElectronicsScrap],
            item 1 [Base.ElectricWire],
            item 1 [Base.SmallSheetMetal],
            item 1 [Base.Screws],
        }
        outputs
        {
            item 1 CeroSec.ElectricStrike,
        }
    }

    /* The one with a motor in it, which is the one part of this a survivor
       cannot wind himself: Base.EngineParts is what the game already calls a
       box of motor and gearing, and it is what a garage door opener was made of
       in 1993 too. */
    craftRecipe MakeCeroSecDoorOperator
    {
        timedAction = MakingElectrical,
        time = 200,
        NeedToBeLearn = true,
        SkillRequired = Electricity:3,
        Tags = AnySurfaceCraft,
        category = Electrical,
        AutoLearnAll = Electricity:9,
        xpAward = Electricity:14,
        inputs
        {
            item 1 tags[base:screwdriver] mode:keep flags[Prop1],
            item 1 tags[base:pliers] mode:keep flags[MayDegradeVeryLight],
            item 1 [Base.EngineParts],
            item 2 [Base.ElectronicsScrap],
            item 2 [Base.ElectricWire],
            item 1 [Base.ScrapMetal],
            item 1 [Base.Screws],
        }
        outputs
        {
            item 1 CeroSec.DoorOperator,
        }
    }

    /* WHERE A SMALL MOTOR COMES FROM: a screwdriver and something that already
       has one in it.

       Build 42 ships no motor item at all (items_cerosec.txt, item SmallMotor),
       so the part has to come out of the world rather than off a bench, and the
       shape of "take a screwdriver to a thing and keep what is inside it" is
       vanilla's own DismantleElectronics
       (media/scripts/generated/recipes/recipes_electrical.txt:37-54): a
       DismantleElectrical timed action, Tags = InHandCraft;Electrical so it is
       done standing up with no table, category = Electrical, xpAward =
       Electricity:2, a screwdriver asked for by tag and KEPT with
       flags[NoBrokenItems], and the thing itself mode:destroy with
       flags[ItemCount;ResearchInput]. Every one of those keys is copied from
       that recipe and from DismantleMiscElectronics (:56-81) beside it.

       WHICH ITEMS. Four, and they are the four things in this game that really
       do have a motor in them (docs/notes/actuators.md, "Which vanilla items
       actually have a motor in them"): Base.HairDryer (a fan),
       Base.SheepElectricShears (clippers), Base.CDplayer (a spindle motor and a
       sled motor) and Base.BlowerFan (a fan with nothing to do). The toys are a
       dead end and the note says why: B42's toys are DisplayCategory = Memento
       keepsakes, and the two that carry base:hasmetal -- ToyCar and ToyPlane --
       are diecast and wind-up, 0.1 weight, no Electric material and no
       electronics tag. A recipe that got a motor out of a teddy bear would be
       this mod inventing a Build 42 that does not exist.

       TWO RECIPES AND NOT ONE, and the reason is the promise: a survivor must
       never be worse off dismantling with ours than with vanilla's.

         Base.HairDryer and Base.SheepElectricShears carry base:miscelectronic,
         so vanilla's DismantleElectronics already takes them and pays ONE
         Base.ElectronicsScrap. Base.BlowerFan carries base:blowerfan, a tag
         used nowhere else in media/scripts, so vanilla dismantles it not at
         all. Ours pays that same scrap and the motor beside it: strictly more,
         for all three.

         Base.CDplayer is different, and it is the caveat the study wrote down.
         It is already an input to DismantleMiscElectronics, whose outputs are
         one Base.ElectronicsScrap plus a second item chosen by an itemMapper --
         and the CD player's mapping is Base.ElectronicsScrap again, so vanilla
         pays TWO. A single recipe cannot pay one item for three inputs and two
         for the fourth, and the mapper cannot do it either: it is keyed by the
         OUTPUT (`Base.Amplifier = Base.Speaker`), so three inputs that all
         yield scrap would collide on one key. So the CD player has its own
         recipe paying two scrap, and a survivor who chooses ours over vanilla's
         gets exactly what vanilla would have given him and a motor.

       LEARNED, like the other four, and from the same book: NeedToBeLearn with
       both names in CeroSec.WiringGuide's LearnedRecipes. A survivor with the
       guide can fit an operator and make one; a survivor without it cannot do
       either, which is the whole point of the guide.

       No SkillRequired, because vanilla's three dismantle recipes have none --
       taking the back off a hair dryer is not a trade. AutoLearnAll =
       Electricity:3 is the one gate: a working electrician eventually notices
       there is a motor in there without being told. */

    craftRecipe DismantleCeroSecMotorAppliance
    {
        timedAction = DismantleElectrical,
        time = 60,
        NeedToBeLearn = true,
        Tags = InHandCraft;Electrical,
        category = Electrical,
        AutoLearnAll = Electricity:3,
        xpAward = Electricity:2,
        inputs
        {
            item 1 tags[base:screwdriver] mode:keep flags[NoBrokenItems],
            item 1 [Base.HairDryer;Base.SheepElectricShears;Base.BlowerFan] mode:destroy flags[ItemCount;ResearchInput],
        }
        outputs
        {
            item 1 CeroSec.SmallMotor,
            item 1 Base.ElectronicsScrap,
        }
    }

    /* The CD player on its own, paying the two scrap DismantleMiscElectronics
       pays, so that choosing this recipe over vanilla's costs nothing. */
    craftRecipe DismantleCeroSecMotorCDPlayer
    {
        timedAction = DismantleElectrical,
        time = 60,
        NeedToBeLearn = true,
        Tags = InHandCraft;Electrical,
        category = Electrical,
        AutoLearnAll = Electricity:3,
        xpAward = Electricity:2,
        inputs
        {
            item 1 tags[base:screwdriver] mode:keep flags[NoBrokenItems],
            item 1 [Base.CDplayer] mode:destroy flags[ItemCount;ResearchInput],
        }
        outputs
        {
            item 1 CeroSec.SmallMotor,
            item 2 Base.ElectronicsScrap,
        }
    }
}
